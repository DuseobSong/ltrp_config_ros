[INSTALL MYSQL-SERVER]
$ sudo apt update
$ sudo apt install mysql-server ufw
$ sudo ufw allow mysql
$ sudo systemctl start mysql
$ sudo systemctl enable mysql

[ROOT ACCOUNT SETTING]
$ sudo mysql -u root
mysql> ALTER USER 'root'@'localhost' IDENTIFIED BY '{PASSWORD}';
mysql> FLUSH PRIVILEGES;

[CHANGE WORKING DATABASE]
mysql> use {DATABASE_NAME};

[GET ALL uSER INFO]
mysql> select Host,User,plugin,authentication_string from mysql.user;

[CREATE USER]
mysql> CREATE USER '{USER_NAME}'@'%' identified by '{PASSWORD}'; # external access
mysql> CREATE uSER '{USER_NAME}'@'localhost' identified by '{PASSWORD}'; # internal access

[CREATE DATABASE]
mysql> CREATE DATABASE {DATABASE_NAME} DEFAULT CHARACTER SET UTF8;

[GET DATABASE LIST]
mysql> SHOW databases;

[CREATE TABLE]
mysql> CREATE TABLE {TABLE_NAME} (
    _id INT PRIMARY KEY AUTO_INCREMENT,
    datetime DATETIME NOT NULL,
    cam_dir VARCHAR(1) NOT NULL,
    width INT UNSIGNED NOT NULL,
    height INT UNSIGNED NOT NULL,
    img_str MEDIUMBLOB NOT NULL
);

[TABLE SUMMARY]
mysql> describe {TABLE_NAME};
